# PRONT 1: OPTIMIZACIONES AGRESIVAS PARA TURSO - SCROLL INFINITO DEL MERCADO

## Contexto y Objetivo

Implementar optimizaciones agresivas de caché exclusivamente para el scroll infinito del mercado, manteniendo el resto del sistema (tiendas, paneles, gift cards, perfil) con actualizaciones inmediatas. Se requiere un balance entre rendimiento (10-60 segundos de desfase para nuevos productos en el mercado) y experiencia de usuario (botón de refresco manual para ver cambios inmediatos).

```typescript
// Caché de 10 segundos para el scroll infinito del mercado
// Los nuevos productos se reflejan en 10-60 segundos máximo
const getMarketProductsCached = unstable_cache(
  async (cursor?: string, limit: number = 12, category?: string, search?: string) => {
    console.log(`[Cache Miss] Consultando Turso para cursor: ${cursor || 'inicio'}`);
    
    const conditions = [];

    if (category && category !== "todos") {
      conditions.push(eq(products.category, category));
    }

    if (search && search.trim()) {
      const searchTerm = `%${search.trim()}%`;
      conditions.push(
        sql`${products.name} LIKE ${searchTerm} OR ${products.description} LIKE ${searchTerm}`
      );
    }

    // Solo productos publicados
    conditions.push(eq(products.isPublished, true));

    if (cursor) {
      conditions.push(sql`${products.id} > ${cursor}`);
    }

    const items = await db
      .select({
        id: products.id,
        name: products.name,
        description: products.description,
        price: products.price,
        stock: products.stock,
        category: products.category,
        imageUrls: products.imageUrls,
        hasVideo: products.hasVideo,
        views: products.views,
        sales: products.sales,
        createdAt: products.createdAt,
        storeId: products.storeId,
      })
      .from(products)
      .where(and(...conditions))
      .orderBy(desc(products.createdAt))
      .limit(limit + 1)
      .all();

    // Obtener nombres de tiendas
    const itemsWithStore = await Promise.all(
      items.map(async (item) => {
        const store = await db
          .select({ name: stores.name, verified: stores.verified })
          .from(stores)
          .where(eq(stores.id, item.storeId))
          .get();
        return {
          ...item,
          storeName: store?.name || "Tienda",
          storeVerified: store?.verified || false,
        };
      })
    );

    const hasMore = itemsWithStore.length > limit;
    const nextCursor = hasMore ? itemsWithStore[limit - 1]?.id : null;

    return {
      items: itemsWithStore.slice(0, limit),
      nextCursor,
      hasMore,
    };
  },
  ["market-products"],
  {
    revalidate: 10, // 10 segundos de caché (máximo 60 segundos de desfase)
    tags: ["market-feed"], // Tag para invalidación manual
  }
);

export async function updateProduct(productId: string, formData: FormData) {
  const user = await requireRole("seller");

  const existingProduct = await db
    .select()
    .from(products)
    .where(eq(products.id, productId))
    .get();

  if (!existingProduct) {
    throw new Error("Producto no encontrado");
  }

  const store = await db
    .select()
    .from(stores)
    .where(eq(stores.userId, user.id))
    .get();

  if (!store || existingProduct.storeId !== store.id) {
    throw new Error("No tienes permiso para editar este producto");
  }

  const rawPrice = formData.get("price");
  const rawStock = formData.get("stock");

  const updateData: any = {};

  const nameValue = formData.get("name");
  if (nameValue) updateData.name = nameValue;

  const descriptionValue = formData.get("description");
  if (descriptionValue) updateData.description = descriptionValue;

  if (rawPrice) updateData.price = parseFloat(rawPrice as string);
  if (rawStock) updateData.stock = parseInt(rawStock as string);

  const categoryValue = formData.get("category");
  if (categoryValue) updateData.category = categoryValue;

  updateData.updatedAt = new Date();

  await db
    .update(products)
    .set(updateData)
    .where(eq(products.id, productId));

  // INVALIDAR CACHÉS AFECTADAS
  invalidateProductCache(store.id);
  
  revalidatePath(`/dashboard/productos/${productId}/editar`);
  revalidatePath("/dashboard/productos");
  redirect("/dashboard/productos?updated=true");
}

  // INVALIDAR CACHÉS AFECTADAS

  redirect("/dashboard/productos?deleted=true");
```

### 2. `components/marketplace/refresh-button.tsx` - Botón de refresco manual

```tsx
"use client";

import { useState } from "react";
import { Button } from "@/components/ui/button";
import { RefreshCw } from "lucide-react";
import { toast } from "sonner";
import { refreshMarketFeed } from "@/app/actions/products";

interface RefreshButtonProps {
  onRefresh?: () => void;
}

export function RefreshButton({ onRefresh }: RefreshButtonProps) {
  const [isRefreshing, setIsRefreshing] = useState(false);

  const handleRefresh = async () => {
    setIsRefreshing(true);
    try {
      await refreshMarketFeed();
      if (onRefresh) onRefresh();
      toast.success("Feed actualizado. Los nuevos productos aparecerán en segundos.");
    } catch (error) {
      toast.error("Error al actualizar el feed");
    } finally {
      setIsRefreshing(false);
    }
  };

  return (
    <Button
      variant="ghost"
      size="sm"
      onClick={handleRefresh}
      disabled={isRefreshing}
      className="gap-2"
    >
      <RefreshCw className={`w-4 h-4 ${isRefreshing ? "animate-spin" : ""}`} />
      {isRefreshing ? "Actualizando..." : "Actualizar productos"}
    </Button>
  );
}
```

## Contexto y Objetivo

### 1. `app/actions/monitoring.ts` - Acciones de monitoreo

```typescript
"use server";

import { db } from "@/db";
import { sql } from "drizzle-orm";
import { requireRole } from "./auth";
import { redis } from "@/lib/redis";

// Tipos
export interface UsageMetrics {
  timestamp: Date;
  activeUsers: number;
  tursoConnections: number;
  tursoSyncGB: number;
  scrollRequests: number;
}

export interface PeakHour {
  hour: number;
  dayOfWeek: string;
  averageUsers: number;
  maxUsers: number;
}

// ============================================
// REGISTRAR MÉTRICAS DE USUARIO
// ============================================

export async function recordUserActivity(userId: string) {
  try {
    const now = new Date();
    const hour = now.getHours();
    const dayOfWeek = now.toLocaleDateString('es-BO', { weekday: 'long' });
    
    // Registrar en Redis para monitoreo en tiempo real
    await redis.hincrby(`stats:users:hour:${hour}`, 'count', 1);
    await redis.hincrby(`stats:users:day:${dayOfWeek}`, 'count', 1);
    
    // Registrar timestamp para sesiones activas
    await redis.zadd('active:sessions', now.getTime(), userId);
    await redis.expire('active:sessions', 3600); // 1 hora
    
  } catch (error) {
    console.error("Error recording user activity:", error);
  }
}

// ============================================
// OBTENER ESTADÍSTICAS DE USO DE TURSO
// ============================================

export async function getTursoMetrics() {
  await requireRole("superadmin");
  
  // Simular métricas de Turso (en producción usar API real)
  // Para demo, usamos valores basados en tiempo real
  
  const now = new Date();
  const hourOfDay = now.getHours();
  
  // Simular picos según hora del día
  let baseConnections = 50;
  if (hourOfDay >= 10 && hourOfDay <= 12) baseConnections = 80;
  if (hourOfDay >= 18 && hourOfDay <= 21) baseConnections = 95;
  if (hourOfDay >= 22) baseConnections = 30;
  
  // Obtener conexiones activas reales de Redis
  const activeSessions = await redis.zcard('active:sessions');
  const activeConnections = Math.max(baseConnections, activeSessions);
  
  // Obtener scroll requests de la última hora
  const scrollRequests = await redis.get('stats:scroll:requests') || 0;
  
  // Simular sincronizaciones (en GB)
  const syncGB = (activeConnections * 0.05) + (scrollRequests * 0.001);
  
  return {
    connections: activeConnections,
    syncGB: parseFloat(syncGB.toFixed(2)),
    syncLimit: 3,
    connectionsLimit: 100,
    scrollRequestsLastHour: parseInt(scrollRequests as string),
    timestamp: now.toISOString(),
  };
}

  const daysOfWeek = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
  
  for (let hour = 0; hour < 24; hour++) {
    const hourData = await redis.hgetall(`stats:users:hour:${hour}`);
    const totalUsers = hourData?.count ? parseInt(hourData.count as string) : 0;
    
    if (totalUsers > 0) {
      peakHours.push({
        hour,
        dayOfWeek: 'Promedio',
        averageUsers: totalUsers,
        maxUsers: totalUsers,
      });
    }
  }
  
  return peakHours.sort((a, b) => b.averageUsers - a.averageUsers).slice(0, 5);
}

// ============================================
// OBTENER CONFIGURACIÓN ACTUAL DEL SCROLL MODE
// ============================================

export async function getScrollMode() {
  const mode = await redis.get('scroll:mode');
  return { mode: mode || 'turso' };
}

// ============================================
// CAMBIAR MODO DE SCROLL (TURSO ↔ XATA)
// ============================================

export async function setScrollMode(mode: 'turso' | 'xata') {
  await requireRole("superadmin");
  
  await redis.set('scroll:mode', mode);
  
  // Registrar la acción en logs
  console.log(`[MONITORING] Scroll mode changed to: ${mode} at ${new Date().toISOString()}`);
  
  return { success: true, mode };
}

// ============================================
// OBTENER ESTADÍSTICAS HISTÓRICAS
// ============================================

export async function getHistoricalStats(days: number = 7) {
  await requireRole("superadmin");
  
  const stats = [];
  for (let i = 0; i < days; i++) {
    const date = new Date();
    date.setDate(date.getDate() - i);
    const dateStr = date.toISOString().split('T')[0];
    
    const dailyUsers = await redis.get(`stats:daily:users:${dateStr}`) || 0;
    const dailyScrolls = await redis.get(`stats:daily:scrolls:${dateStr}`) || 0;
    
    stats.push({
      date: dateStr,
      users: parseInt(dailyUsers as string),
      scrolls: parseInt(dailyScrolls as string),
    });
  }
  
  return stats.reverse();
}
```
